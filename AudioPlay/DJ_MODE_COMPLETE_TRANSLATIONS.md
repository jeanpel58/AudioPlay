# Traductions complètes FormDJ - Toutes les langues

## Clés de traduction à ajouter dans tous les fichiers de ressources

### 🇫🇷 Français (Resources.resx)

```xml
  <!-- Messages d'erreur -->
  <data name="DJ_Error_LoadingDeck" xml:space="preserve">
	<value>Erreur chargement Deck {0} : {1}</value>
  </data>
  <data name="DJ_Error_Title" xml:space="preserve">
	<value>Erreur</value>
  </data>

  <!-- Messages SYNC -->
  <data name="DJ_BPM_NotDetected" xml:space="preserve">
	<value>BPM non détecté sur les deux platines.</value>
  </data>
  <data name="DJ_Sync_TitleAtoB" xml:space="preserve">
	<value>Sync A → B</value>
  </data>
  <data name="DJ_Sync_TitleBtoA" xml:space="preserve">
	<value>Sync B → A</value>
  </data>

  <!-- Labels dynamiques -->
  <data name="DJ_BPM_Unknown" xml:space="preserve">
	<value>BPM: --</value>
  </data>
  <data name="DJ_BPM_Value" xml:space="preserve">
	<value>BPM: {0:F1}</value>
  </data>
  <data name="DJ_Pitch_Value" xml:space="preserve">
	<value>Pitch: {0:+0.0%;-0.0%;0.0%}</value>
  </data>
  <data name="DJ_Duration_Format" xml:space="preserve">
	<value>{0:mm\:ss} / {1:mm\:ss}</value>
  </data>

  <!-- Boutons Play/Pause (symboles universels) -->
  <data name="DJ_Button_Play" xml:space="preserve">
	<value>▶</value>
  </data>
  <data name="DJ_Button_Pause" xml:space="preserve">
	<value>⏸</value>
  </data>
  <data name="DJ_Button_Stop" xml:space="preserve">
	<value>⏹</value>
  </data>
  <data name="DJ_Button_Cue" xml:space="preserve">
	<value>CUE</value>
  </data>
```
